SUPABASE PROJECT SETUP
==============================================================

1. Create a Supabase project at https://supabase.com
2. Note your project URL and anon key (public key only)
3. Add to .env.local:
     VITE_SUPABASE_URL=https://your-project.supabase.co
     VITE_SUPABASE_ANON_KEY=eyJ...your-anon-key...
4. NEVER add SUPABASE_SERVICE_ROLE_KEY to frontend code
5. Enable Email Auth (or SSO/Magic Link) in Supabase Auth settings
6. Run the SQL schema below in Supabase SQL editor
7. Enable RLS on all tables (see SQL below)
8. Test connection via Demo/Live Settings → Backend Setup

==============================================================
TRIGGERS — updated_at auto-update
==============================================================

CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Apply trigger to all tables with updated_at
DO $$
DECLARE
  t TEXT;
BEGIN
  FOREACH t IN ARRAY ARRAY[
    'organisations','responders','service_users','missions',
    'welfare_visits','check_ins','risk_flags','incidents',
    'messages','evidence_records','sync_events','app_settings'
  ] LOOP
    EXECUTE FORMAT('
      DROP TRIGGER IF EXISTS trg_updated_at ON %I;
      CREATE TRIGGER trg_updated_at
        BEFORE UPDATE ON %I
        FOR EACH ROW EXECUTE FUNCTION update_updated_at();
    ', t, t);
  END LOOP;
END;
$$;

RLS POLICIES (PLACEHOLDER — ADAPT TO YOUR ORGANISATION)
==============================================================

-- These are example policies only. Final access control
-- design must be reviewed by your organisation and DPO
-- before live deployment.

-- Authenticated users can read their organisation's records
CREATE POLICY "org_read_missions" ON missions
  FOR SELECT USING (auth.uid() IS NOT NULL);

CREATE POLICY "org_insert_missions" ON missions
  FOR INSERT WITH CHECK (auth.uid() IS NOT NULL);

CREATE POLICY "org_update_missions" ON missions
  FOR UPDATE USING (auth.uid() IS NOT NULL);

-- Supervisor reviews are append-only (no update/delete)
CREATE POLICY "read_supervisor_reviews" ON supervisor_reviews
  FOR SELECT USING (auth.uid() IS NOT NULL);

CREATE POLICY "insert_supervisor_reviews" ON supervisor_reviews
  FOR INSERT WITH CHECK (auth.uid() IS NOT NULL);

-- Add equivalent policies for all other tables.

-- Engage your DPO and legal team before deploying with real data.

-- NOTE: DO NOT add policies that expose service_users data
-- to unauthenticated (anon) users. Service user data is
-- sensitive and must only be accessible to authenticated
-- organisation staff with appropriate roles.
